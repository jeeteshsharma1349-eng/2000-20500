E) Replace recalc() with this version
-------------------------------------
function recalc(){
 let balance=config.start, dayRows=[],cumPL=0;
 for(let d=1;d<=config.days;d++){
  const opening=balance, goal=planned()[d-1].goal;
  const stopFloor=opening*(1-(Number(config.stopLossRate??24)/100));
  const list=getTrades(d);
  let wins=0,losses=0,pending=0,dayPL=0,stopHit=false,blocked=0;

  list.forEach(t=>{
   if(t.result==='PENDING'){pending++;return}
   if(stopHit){blocked++;return}
   if(t.result==='WIN'){wins++}else{losses++}
   const pl=t.result==='WIN'?t.stake*config.payout/100:-t.stake;
   t.pl=pl;
   balance+=pl;
   dayPL+=pl;
   cumPL+=pl;
   t.balanceAfter=balance;
   if(balance<=stopFloor) stopHit=true;
  });

  const pct=goal>opening?Math.max(0,Math.min(100,(balance-opening)/(goal-opening)*100)):(balance>=goal?100:0);
  let status='PENDING';
  if(balance>=goal)status='COMPLETE';
  else if(list.length||d===selectedDay)status='PROGRESS';

  dayRows.push({
   day:d,opening,goal,actual:balance,dayPL,wins,losses,pending,count:list.length,
   pct,status,list,stopFloor,stopHit,blocked
  });
 }
 return {dayRows,balance,profit:balance-config.start};
}

G) Add a day-lock helper and enforce it
---------------------------------------
Add this function after liveBalance():

function isDayStopLocked(day){
 const row=recalc().dayRows[day-1];
 return !!(row && row.stopHit);
}

At the very beginning of addTrade(), after reading day, add:
if(isDayStopLocked(day)){
  toast('Daily stop loss hit. No more trades can be added today.');
  return;
}

At the very beginning of resolveTrade(id,result), after finding t, add:
if(t && isDayStopLocked(t.day)){
  toast('Daily stop loss hit. This day is locked; pending trades cannot be resolved here.');
  return;
}
